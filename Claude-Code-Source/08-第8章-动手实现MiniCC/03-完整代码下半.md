## 1️⃣ Permission 过滤 + 检查

```python
def filter_tools_by_mode(mode: PermissionMode) -> list[Tool]:
    """Plan Mode 只返回 read-only 工具。"""
    if mode == PermissionMode.PLAN:
        return [t for t in ALL_TOOLS if t.is_read_only]
    return ALL_TOOLS


def check_permission(tool: Tool, args: dict, mode: PermissionMode) -> tuple[bool, str]:
    """返回 (allowed, reason)."""
    # 第 0 层：危险检测
    if tool.name == "Bash":
        d = check_dangerous(args.get("command", ""))
        if d:
            return False, f"BLOCKED dangerous: {d}"

    # 第 1 层：Mode 检查
    if mode == PermissionMode.PLAN and not tool.is_read_only:
        return False, f"Plan Mode: {tool.name} is not read-only"
    if mode == PermissionMode.BYPASS:
        return True, ""
    if mode == PermissionMode.ACCEPT_EDITS and tool.name in ("Edit", "Write"):
        return True, ""

    # 第 4 层：destructive 弹窗
    if tool.is_destructive:
        console.print(Panel(
            f"[bold yellow]Tool:[/] {tool.name}\n"
            f"[bold yellow]Input:[/] {json.dumps(args, ensure_ascii=False, indent=2)}",
            title="⚠️  Permission Request",
            border_style="yellow",
        ))
        if Confirm.ask("Allow this operation?", default=False):
            return True, ""
        return False, "User denied"

    return True, ""
```

## 2️⃣ System Prompt 构建

```python
def build_system_prompt(mode: PermissionMode) -> str:
    sections = [
        # 1. 身份
        "You are MiniCC, an interactive CLI coding agent. "
        "You help users with software engineering tasks via the available tools.",
        "",
        # 4. 语气
        "# Tone and style",
        " - Be concise. Output text to communicate with the user.",
        " - Reference code with `file_path:line_number` format.",
        " - Don't use emojis unless explicitly asked.",
        " - Don't include a colon before tool calls.",
        "",
        # 6. 做任务
        "# Doing tasks",
        " - Read files before modifying them.",
        " - Don't add features beyond what was asked.",
        " - Before claiming task complete, verify it actually works.",
        " - If a tool call is denied, think about WHY and adjust approach.",
        " - Make multiple independent tool calls in parallel when possible.",
        "",
        # 8. 工具使用
        "# Using your tools",
        " - Read: read files (preferred over `cat`)",
        " - Write: create new files (DESTRUCTIVE)",
        " - Edit: replace exact strings (use Read first to see exact text)",
        " - Bash: run shell commands (DANGEROUS)",
        " - Glob: find files by pattern",
        "",
        # 安全
        "# Safety",
        " - Dangerous commands (rm -rf /, mkfs, fork bombs) are auto-blocked.",
        f" - Current permission mode: {mode.value}",
    ]
    if mode == PermissionMode.PLAN:
        sections.append(
            " - PLAN MODE: you cannot modify anything. "
            "Discuss approach first, then ask user to exit Plan Mode."
        )
    sections += [
        "",
        # 10. 环境
        "# Environment",
        f" - CWD: {os.getcwd()}",
        f" - Platform: {sys.platform}",
    ]
    return "\n".join(sections)
```

## 3️⃣ Auto-Compact

```python
def estimate_tokens(messages: list) -> int:
    try:
        s = json.dumps(messages, ensure_ascii=False, default=str)
    except Exception:
        s = str(messages)
    cn = sum(1 for c in s if "\u4e00" <= c <= "\u9fff")
    return cn * 2 + (len(s) - cn) // 4


async def auto_compact(client: Anthropic, messages: list) -> list:
    console.print("[yellow]⚠️  Triggering auto-compact...[/]")
    summary_prompt = (
        "CRITICAL: Respond with TEXT ONLY. Do NOT call any tools.\n\n"
        "Summarize the conversation above. Preserve:\n"
        "- User's original intent\n"
        "- Key technical decisions and file paths\n"
        "- Important error messages and how they were resolved\n"
        "- Any pending todos\n\n"
        "Wrap your summary in <summary></summary> tags."
    )
    convo_text = "\n".join(
        f"[{m['role']}] " + (
            m["content"] if isinstance(m["content"], str)
            else json.dumps(m["content"], ensure_ascii=False, default=str)[:500]
        )
        for m in messages
    )
    resp = client.messages.create(
        model=MODEL,
        max_tokens=2000,
        messages=[{
            "role": "user",
            "content": f"Conversation so far:\n\n{convo_text}\n\n{summary_prompt}"
        }],
    )
    summary = resp.content[0].text
    console.print(f"[green]✓ Compacted to {len(summary)} chars[/]")
    return [{"role": "user", "content": f"[Compacted summary of prior conversation]\n{summary}"}]
```

## 4️⃣ 工具结果转 dict 辅助

```python
def blocks_to_dict(content) -> list:
    """Convert SDK content blocks to JSON-serialisable dicts."""
    result = []
    for block in content:
        if block.type == "text":
            result.append({"type": "text", "text": block.text})
        elif block.type == "tool_use":
            result.append({
                "type": "tool_use",
                "id": block.id,
                "name": block.name,
                "input": block.input,
            })
    return result
```

**为啥需要**：Anthropic SDK 返回的 content 是对象。要塞回 messages 给下一次请求时，必须转成纯 dict（否则不能 JSON 序列化）。

## 5️⃣ ⭐ Agent 主循环

```python
async def run_agent(client: Anthropic, user_input: str, messages: list, mode: PermissionMode) -> list:
    messages.append({"role": "user", "content": user_input})

    for turn in range(1, MAX_TURNS + 1):
        # === 护栏 1: auto-compact 检查 ===
        if estimate_tokens(messages) > AUTO_COMPACT_THRESHOLD:
            messages = await auto_compact(client, messages)

        # === 准备工具列表（按 mode 过滤）===
        tools_for_api = [t.to_api_spec() for t in filter_tools_by_mode(mode)]

        # === A. 调 Claude API ===
        console.print(f"[dim]── turn {turn} ──[/]")
        try:
            response = client.messages.create(
                model=MODEL,
                max_tokens=MAX_TOKENS,
                system=build_system_prompt(mode),
                tools=tools_for_api,
                messages=messages,
            )
        except Exception as e:
            console.print(f"[red]API error: {e}[/]")
            return messages

        # 收集 assistant 的回复
        text_parts = []
        tool_uses = []
        for block in response.content:
            if block.type == "text":
                text_parts.append(block.text)
            elif block.type == "tool_use":
                tool_uses.append(block)

        # 显示文本
        if text_parts:
            text = "\n".join(text_parts)
            console.print(Panel(Markdown(text), title="🤖 Assistant", border_style="blue"))

        # 存 assistant 回复
        messages.append({"role": "assistant", "content": blocks_to_dict(response.content)})

        # === B. 没有工具 = 退出 ===
        if not tool_uses:
            return messages

        # === C. 执行所有 tool_use ===
        tool_results = []
        for tu in tool_uses:
            tool = TOOLS_BY_NAME.get(tu.name)
            if not tool:
                tool_results.append({
                    "type": "tool_result", "tool_use_id": tu.id,
                    "content": f"ERROR: unknown tool {tu.name}", "is_error": True
                })
                continue

            args_preview = json.dumps(tu.input, ensure_ascii=False)[:80]
            console.print(f"[cyan]→ {tu.name}({args_preview})[/]")

            # 权限检查
            allowed, reason = check_permission(tool, tu.input, mode)
            if not allowed:
                console.print(f"  [red]✗ {reason}[/]")
                tool_results.append({
                    "type": "tool_result", "tool_use_id": tu.id,
                    "content": reason, "is_error": True
                })
                continue

            # 执行
            try:
                result = await tool.handler(**tu.input)
            except TypeError as e:
                result = f"ERROR: bad arguments: {e}"
            except Exception as e:
                result = f"ERROR: {e}"

            # 截断防爆
            if len(result) > MAX_TOOL_RESULT_CHARS:
                result = result[:MAX_TOOL_RESULT_CHARS] + f"\n... (truncated, {len(result)} total chars)"

            preview = result.split("\n")[0][:80]
            console.print(f"  [green]✓[/] {preview}")
            tool_results.append({
                "type": "tool_result", "tool_use_id": tu.id, "content": result
            })

        # === D. 塞回 messages（注意 user 角色！）===
        messages.append({"role": "user", "content": tool_results})

    # === 护栏 2: 轮次上限 ===
    console.print(f"[yellow]Reached max_turns ({MAX_TURNS})[/]")
    return messages
```

## 6️⃣ CLI 入口（REPL）

```python
def print_help():
    console.print(Panel("""
[bold]MiniCC commands:[/]
  /help     show this help
  /mode     show current permission mode
  /plan     switch to Plan Mode (read-only)
  /default  switch to Default Mode
  /accept   switch to Accept Edits Mode
  /bypass   switch to Bypass Mode (⚠️  DANGEROUS)
  /clear    clear conversation history
  /tokens   show estimated token usage
  /exit     quit
    """, border_style="cyan"))


async def main():
    if not os.environ.get("ANTHROPIC_API_KEY"):
        console.print("[red]Set ANTHROPIC_API_KEY env var first.[/]")
        sys.exit(1)

    client = Anthropic()
    messages: list = []
    mode = PermissionMode.DEFAULT

    console.print(Panel.fit(
        "[bold cyan]MiniCC[/] — a Mini Claude Code in Python\n"
        f"Model: {MODEL}\n"
        "Type /help for commands, /exit to quit.",
        border_style="cyan",
    ))

    while True:
        try:
            user_input = console.input(f"\n[bold green]{mode.value}>[/] ").strip()
        except (KeyboardInterrupt, EOFError):
            console.print("\n[dim]bye[/]")
            break

        if not user_input:
            continue

        if user_input.startswith("/"):
            cmd = user_input.lower()
            if cmd in ("/exit", "/quit"):
                break
            elif cmd == "/help":
                print_help()
            elif cmd == "/mode":
                console.print(f"Current mode: [yellow]{mode.value}[/]")
            elif cmd == "/plan":
                mode = PermissionMode.PLAN
                console.print("[yellow]→ Plan Mode (read-only)[/]")
            elif cmd == "/default":
                mode = PermissionMode.DEFAULT
                console.print("[green]→ Default Mode[/]")
            elif cmd == "/accept":
                mode = PermissionMode.ACCEPT_EDITS
                console.print("[green]→ Accept Edits Mode[/]")
            elif cmd == "/bypass":
                if Confirm.ask("[red]⚠️  Bypass Mode disables permission prompts. Continue?[/]", default=False):
                    mode = PermissionMode.BYPASS
                    console.print("[red]→ Bypass Mode (dangerous patterns still blocked)[/]")
            elif cmd == "/clear":
                messages = []
                console.print("[dim]History cleared[/]")
            elif cmd == "/tokens":
                console.print(f"Estimated tokens: [yellow]{estimate_tokens(messages)}[/]")
            else:
                console.print(f"[red]Unknown command: {cmd}[/]")
            continue

        messages = await run_agent(client, user_input, messages, mode)


if __name__ == "__main__":
    asyncio.run(main())
```
