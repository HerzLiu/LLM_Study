## 2️⃣ Permission Mode + 危险检测

```python
# Permission Mode（第 7 章）
class PermissionMode(Enum):
    PLAN = "plan"
    DEFAULT = "default"
    ACCEPT_EDITS = "acceptEdits"
    BYPASS = "bypass"

# 危险命令模式（第 7 章 7.3）
DANGEROUS_PATTERNS = [
    (r"\brm\s+-rf\s+/(?:\s|$)",     "rm -rf root"),
    (r"\bmkfs\b",                    "format filesystem"),
    (r":\(\)\s*\{\s*:\|:&\s*\};\s*:", "fork bomb"),
    (r"\bdd\s+.*of=/dev/[sh]d",     "raw disk write"),
    (r">\s*/dev/sd",                 "raw disk redirect"),
    (r"\bchmod\s+-R\s+777\s+/",     "chmod 777 root"),
]

def check_dangerous(cmd: str) -> str | None:
    """返回危险名，安全返回 None。"""
    for pattern, name in DANGEROUS_PATTERNS:
        if re.search(pattern, cmd):
            return name
    return None
```

详见 [Bash 危险模式](../07-%E7%AC%AC7%E7%AB%A0-%E5%AE%89%E5%85%A8%E4%B8%8E%E6%9D%83%E9%99%90/03-Bash%E5%8D%B1%E9%99%A9%E6%A8%A1%E5%BC%8F.md)。

## 3️⃣ Tool 基类

```python
@dataclass
class Tool:
    name: str
    description: str                  # 给 UI
    input_schema: dict                # JSON Schema
    handler: Callable                 # async 函数：(args) -> str
    is_read_only: bool = False
    is_destructive: bool = False
    is_concurrency_safe: bool = True

    def to_api_spec(self) -> dict:
        """转成 Anthropic API 期望的 tool 定义格式。"""
        return {
            "name": self.name,
            "description": self.description,
            "input_schema": self.input_schema,
        }
```

简化版（vs 第 3 章 `Tool` 接口的 18 字段）：
- 没分 `description` 和 `prompt` —— 直接用 description 当 prompt
- 没有 `preparePermissionMatcher` 等高级字段

## 4️⃣ Read 工具

```python
async def read_handler(file_path: str, offset: int = 1, limit: int = 2000) -> str:
    p = Path(file_path).expanduser()
    if not p.is_absolute():
        return f"ERROR: path must be absolute, got: {file_path}"
    if not p.exists():
        return f"ERROR: file not found: {p}"
    if p.is_dir():
        return f"ERROR: is a directory, use Bash `ls` instead"
    try:
        lines = p.read_text(encoding="utf-8", errors="replace").splitlines()
    except Exception as e:
        return f"ERROR: {e}"
    offset = max(1, int(offset))
    limit = max(1, int(limit))
    end = min(offset - 1 + limit, len(lines))
    selected = lines[offset - 1:end]
    # ⭐ 关键：加行号（cat -n 格式）
    return "\n".join(f"{offset + i}: {line}" for i, line in enumerate(selected))

READ_TOOL = Tool(
    name="Read",
    description=(
        "Read a file from the local filesystem.\n"
        "- file_path must be an absolute path\n"
        "- Results returned with line numbers in `<num>: <content>` format\n"
        "- By default reads up to 2000 lines from the start\n"
        "- Use offset/limit for large files"
    ),
    input_schema={
        "type": "object",
        "properties": {
            "file_path": {"type": "string", "description": "Absolute file path"},
            "offset": {"type": "integer", "description": "Start line (1-indexed)"},
            "limit": {"type": "integer", "description": "Max lines to read"},
        },
        "required": ["file_path"],
    },
    is_read_only=True,
    handler=read_handler,
)
```

**关键设计**（对照 [FileReadTool 解剖](../03-%E7%AC%AC3%E7%AB%A0-%E5%B7%A5%E5%85%B7%E7%B3%BB%E7%BB%9F/03-FileReadTool%E8%A7%A3%E5%89%96.md)）：
- 行号格式 `<num>: <content>` —— Edit 的协同设计
- 路径必须绝对 —— 防止意外读跨目录
- 截 2000 行 —— 防止大文件爆 context
- `is_read_only=True` —— Plan Mode 可用

## 5️⃣ Write 工具

```python
async def write_handler(file_path: str, content: str) -> str:
    p = Path(file_path).expanduser()
    if not p.is_absolute():
        return f"ERROR: path must be absolute, got: {file_path}"
    try:
        p.parent.mkdir(parents=True, exist_ok=True)
        p.write_text(content, encoding="utf-8")
        return f"OK: wrote {len(content)} chars to {p}"
    except Exception as e:
        return f"ERROR: {e}"

WRITE_TOOL = Tool(
    name="Write",
    description="Write content to a file (overwrite). Path must be absolute. Parent dirs auto-created.",
    input_schema={
        "type": "object",
        "properties": {
            "file_path": {"type": "string"},
            "content": {"type": "string"},
        },
        "required": ["file_path", "content"],
    },
    is_destructive=True,                 # ⭐ 触发权限弹窗
    handler=write_handler,
)
```

## 6️⃣ Edit 工具

```python
async def edit_handler(file_path: str, old_string: str, new_string: str) -> str:
    p = Path(file_path).expanduser()
    if not p.is_absolute():
        return "ERROR: path must be absolute"
    if not p.exists():
        return f"ERROR: file not found: {p}"
    content = p.read_text(encoding="utf-8")
    count = content.count(old_string)
    # ⭐ 关键：必须唯一匹配
    if count == 0:
        return "ERROR: old_string not found in file"
    if count > 1:
        return f"ERROR: old_string appears {count} times, must be unique"
    new_content = content.replace(old_string, new_string, 1)
    p.write_text(new_content, encoding="utf-8")
    return f"OK: replaced 1 occurrence in {p}"

EDIT_TOOL = Tool(
    name="Edit",
    description=(
        "Replace exactly one occurrence of old_string with new_string in a file.\n"
        "- old_string MUST be unique in the file (include surrounding context if needed)\n"
        "- Use Read first to see the exact content with line numbers"
    ),
    input_schema={
        "type": "object",
        "properties": {
            "file_path":  {"type": "string"},
            "old_string": {"type": "string"},
            "new_string": {"type": "string"},
        },
        "required": ["file_path", "old_string", "new_string"],
    },
    is_destructive=True,
    handler=edit_handler,
)
```

**关键设计**：**精确字符串匹配 + 唯一性校验**。
- 0 个匹配 → 失败（位置不对）
- > 1 个匹配 → 失败（不唯一，提示加 context）
- 1 个匹配 → 替换

这就是为什么 Read 要带行号——给模型看到完整原文才能写出唯一的 old_string。

## 7️⃣ Bash 工具

```python
async def bash_handler(command: str, timeout: int = 30) -> str:
    # ⭐ 第 0 层防御：危险模式检测
    danger = check_dangerous(command)
    if danger:
        return f"BLOCKED: dangerous pattern detected: {danger}"
    try:
        proc = await asyncio.create_subprocess_shell(
            command,
            stdout=asyncio.subprocess.PIPE,
            stderr=asyncio.subprocess.PIPE,
        )
        stdout, stderr = await asyncio.wait_for(proc.communicate(), timeout=timeout)
    except asyncio.TimeoutError:
        return f"ERROR: command timed out after {timeout}s"
    except Exception as e:
        return f"ERROR: {e}"
    out = stdout.decode("utf-8", errors="replace")
    err = stderr.decode("utf-8", errors="replace")
    return f"[exit={proc.returncode}]\nSTDOUT:\n{out}\nSTDERR:\n{err}"

BASH_TOOL = Tool(
    name="Bash",
    description=(
        "Execute a shell command. Returns stdout/stderr.\n"
        "- Default timeout: 30s\n"
        "- Dangerous patterns (rm -rf /, mkfs, fork bombs) are auto-blocked\n"
        "- Prefer Read/Edit/Write/Glob over cat/sed/echo/find when applicable"
    ),
    input_schema={
        "type": "object",
        "properties": {
            "command": {"type": "string"},
            "timeout": {"type": "integer"},
        },
        "required": ["command"],
    },
    is_destructive=True,
    is_concurrency_safe=False,           # ⭐ Bash 不能乱并发
    handler=bash_handler,
)
```

**3 层防护**：
1. `check_dangerous()` 在 handler 内
2. `is_destructive=True` 触发弹窗
3. `asyncio.wait_for` 限时

## 8️⃣ Glob 工具

```python
async def glob_handler(pattern: str, path: str = ".") -> str:
    p = Path(path).expanduser()
    matches = sorted(glob_module.glob(str(p / pattern), recursive=True))[:200]
    if not matches:
        return f"No matches for pattern: {pattern}"
    return "\n".join(matches)

GLOB_TOOL = Tool(
    name="Glob",
    description="Find files by glob pattern (e.g. '**/*.py'). Returns up to 200 paths.",
    input_schema={
        "type": "object",
        "properties": {
            "pattern": {"type": "string"},
            "path": {"type": "string"},
        },
        "required": ["pattern"],
    },
    is_read_only=True,
    handler=glob_handler,
)
```
